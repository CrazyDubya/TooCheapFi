### 2.2 CSV Streaming Export
**File**: `Sources/HistoryStore.swift:349-365`

**Problem**: Builds entire CSV string in memory (1000 records = 1MB+)

**Solution**: Stream to file handle:

```swift
func exportToCSV(to url: URL) throws {
    let handle = try FileHandle(forWritingTo: url)
    defer { handle.closeFile() }

    handle.write("Timestamp,Interface,...\n".data(using: .utf8)!)

    // Stream each row
    while sqlite3_step(stmt) == SQLITE_ROW {
        let row = buildCSVRow(stmt)
        handle.write(row.data(using: .utf8)!)
    }
}
```

### 4.3 Status Caching
**File**: `Sources/NetworkMonitor.swift`

```swift
private func cacheStatus(_ status: NetworkStatus) {
    let encoder = JSONEncoder()
    if let data = try? encoder.encode(status) {
        try? data.write(to: Preferences.configDirectory.appendingPathComponent("last_status.json"))
    }
}

private func loadCachedStatus() -> NetworkStatus? {
    let path = Preferences.configDirectory.appendingPathComponent("last_status.json")
    guard let data = try? Data(contentsOf: path) else { return nil }
    return try? JSONDecoder().decode(NetworkStatus.self, from: data)
}
```
