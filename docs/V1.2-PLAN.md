## 1. Critical Security Fixes

### 1.1 SQL Injection Prevention
**Files:** `HistoryStore.swift`

| Location | Issue | Fix |
|----------|-------|-----|
| Line 259 | String interpolation in SQL | Use parameterized query with bind |
| Lines 365-367 | Interpolated retention days | Use sqlite3_bind_int for temporal |

**Before:**
```swift
WHERE timestamp > datetime('now', '-\(hours) hours', 'localtime');
```

**After:**
```swift
WHERE timestamp > datetime('now', ? || ' hours', 'localtime');
// sqlite3_bind_text(stmt, 1, "-\(hours)", -1, nil)
```

### 2.2 Thread-Safe Properties
**Files:** `NetworkMonitor.swift`, `NotificationManager.swift`, `Preferences.swift`

Create thread-safe property wrapper:

```swift
@propertyWrapper
struct ThreadSafe<Value> {
    private var value: Value
    private let lock = NSLock()

    var wrappedValue: Value {
        get { lock.withLock { value } }
        set { lock.withLock { value = newValue } }
    }
}
```

Apply to:
- `NetworkMonitor.currentStatus`
- `NetworkMonitor.lastSpeedTest`
- `NotificationManager.lastNotifiedStatus`
- `NotificationManager.outageStartTime`

## 7. Event Hooks System

### 7.1 Hook Protocol
**New File:** `EventHooks.swift`

```swift
/// Protocol for receiving network status events
protocol NetworkEventObserver: AnyObject {
    func onStatusChanged(from: NetworkStatus?, to: NetworkStatus)
    func onOutageStarted(layer: String, time: Date)
    func onOutageEnded(layer: String, duration: TimeInterval)
    func onQualityChanged(from: ConnectionQuality, to: ConnectionQuality)
    func onSpeedTestCompleted(result: SpeedTestResult)
}

/// Extension with default implementations (all optional)
extension NetworkEventObserver {
    func onStatusChanged(from: NetworkStatus?, to: NetworkStatus) {}
    func onOutageStarted(layer: String, time: Date) {}
    func onOutageEnded(layer: String, duration: TimeInterval) {}
    func onQualityChanged(from: ConnectionQuality, to: ConnectionQuality) {}
    func onSpeedTestCompleted(result: SpeedTestResult) {}
}
```

### 7.2 Hook Manager
```swift
class EventHookManager {
    static let shared = EventHookManager()

    private var observers = NSHashTable<AnyObject>.weakObjects()
    private let queue = DispatchQueue(label: "com.toocheapfi.hooks")

    func addObserver(_ observer: NetworkEventObserver) {
        queue.async { self.observers.add(observer) }
    }

    func removeObserver(_ observer: NetworkEventObserver) {
        queue.async { self.observers.remove(observer) }
    }

    func notifyStatusChanged(from: NetworkStatus?, to: NetworkStatus) {
        queue.async {
            for case let observer as NetworkEventObserver in self.observers.allObjects {
                DispatchQueue.main.async {
                    observer.onStatusChanged(from: from, to: to)
                }
            }
        }
    }

    // Similar methods for other events...
}
```

### 7.3 Shell Script Hooks
**New File:** `ScriptHookRunner.swift`

```swift
class ScriptHookRunner: NetworkEventObserver {
    private let scriptsPath: URL

    init() {
        scriptsPath = Preferences.configDirectory.appendingPathComponent("hooks")
        try? FileManager.default.createDirectory(at: scriptsPath, withIntermediateDirectories: true)
    }

    func onOutageStarted(layer: String, time: Date) {
        runScript("on-outage-start", args: [layer, ISO8601DateFormatter().string(from: time)])
    }

    func onOutageEnded(layer: String, duration: TimeInterval) {
        runScript("on-outage-end", args: [layer, String(Int(duration))])
    }

    private func runScript(_ name: String, args: [String]) {
        let scriptPath = scriptsPath.appendingPathComponent(name)
        guard FileManager.default.isExecutableFile(atPath: scriptPath.path) else { return }

        DispatchQueue.global(qos: .utility).async {
            let task = Process()
            task.executableURL = scriptPath
            task.arguments = args
            try? task.run()
        }
    }
}
```
